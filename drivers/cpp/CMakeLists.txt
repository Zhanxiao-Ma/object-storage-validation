cmake_minimum_required(VERSION 3.20)
project(osv_sdk_worker LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(nlohmann_json 3.11 REQUIRED)
option(OSV_BUILD_S3 "Build the AWS S3 worker" ON)
if(OSV_BUILD_S3)
  find_package(AWSSDK REQUIRED COMPONENTS s3)
  add_executable(osv-cpp-s3 worker.cpp)
  target_link_libraries(osv-cpp-s3 PRIVATE ${AWSSDK_LINK_LIBRARIES} nlohmann_json::nlohmann_json)
  target_compile_options(osv-cpp-s3 PRIVATE -Wall -Wextra -Wpedantic)
endif()

# Native SDK sources are supplied separately and retain upstream licenses.
set(OSV_OSS_SOURCE "" CACHE PATH "Path to aliyun/aliyun-oss-cpp-sdk checkout")
set(OSV_TOS_SOURCE "" CACHE PATH "Path to volcengine/ve-tos-cpp-sdk checkout")
if(OSV_OSS_SOURCE OR OSV_TOS_SOURCE)
  find_package(CURL REQUIRED)
  find_package(OpenSSL REQUIRED)
  find_package(Threads REQUIRED)
endif()
if(OSV_OSS_SOURCE)
  set(BUILD_SAMPLE OFF CACHE BOOL "" FORCE)
  set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
  add_subdirectory(${OSV_OSS_SOURCE} oss-sdk)
  file(STRINGS "${OSV_OSS_SOURCE}/VERSION" OSV_OSS_VERSION)
  add_executable(osv-cpp-oss native_worker.cpp)
  target_include_directories(osv-cpp-oss PRIVATE ${OSV_OSS_SOURCE}/sdk/include)
  target_compile_definitions(osv-cpp-oss PRIVATE OSV_OSS OSV_NATIVE_VERSION="${OSV_OSS_VERSION}")
  target_link_libraries(osv-cpp-oss PRIVATE cpp-sdk nlohmann_json::nlohmann_json CURL::libcurl OpenSSL::SSL OpenSSL::Crypto Threads::Threads)
endif()
if(OSV_TOS_SOURCE)
  set(BUILD_UNITTEST OFF CACHE BOOL "" FORCE)
  set(BUILD_DEMO OFF CACHE BOOL "" FORCE)
  add_subdirectory(${OSV_TOS_SOURCE} tos-sdk)
  add_executable(osv-cpp-tos native_worker.cpp)
  target_include_directories(osv-cpp-tos PRIVATE ${OSV_TOS_SOURCE}/sdk/include)
  file(STRINGS "${OSV_TOS_SOURCE}/CMakeLists.txt" OSV_TOS_VERSION_LINE REGEX "VERSION [0-9]+\\.[0-9]+\\.[0-9]+")
  string(REGEX MATCH "[0-9]+\\.[0-9]+\\.[0-9]+" OSV_TOS_VERSION "${OSV_TOS_VERSION_LINE}")
  if(NOT OSV_TOS_VERSION)
    message(FATAL_ERROR "Cannot identify the supplied TOS SDK version")
  endif()
  target_compile_definitions(osv-cpp-tos PRIVATE OSV_NATIVE_VERSION="${OSV_TOS_VERSION}")
  target_link_libraries(osv-cpp-tos PRIVATE ve-tos-cpp-sdk-lib nlohmann_json::nlohmann_json CURL::libcurl OpenSSL::SSL OpenSSL::Crypto Threads::Threads)
endif()

# COS upstream uses CMAKE_SOURCE_DIR internally and must be built as a standalone project.
set(OSV_COS_SOURCE "" CACHE PATH "Official tencentyun/cos-cpp-sdk-v5 checkout")
set(OSV_COS_LIBRARY "" CACHE FILEPATH "Static libcossdk.a built from OSV_COS_SOURCE")
if(OSV_COS_SOURCE OR OSV_COS_LIBRARY)
  if(NOT OSV_COS_SOURCE OR NOT EXISTS "${OSV_COS_LIBRARY}")
    message(FATAL_ERROR "COS requires both OSV_COS_SOURCE and a built OSV_COS_LIBRARY")
  endif()
  find_package(Poco REQUIRED COMPONENTS Foundation Net NetSSL Crypto Util JSON XML)
  find_package(OpenSSL REQUIRED)
  find_package(Threads REQUIRED)
  add_executable(osv-cpp-cos cos_worker.cpp)
  target_include_directories(osv-cpp-cos PRIVATE "${OSV_COS_SOURCE}/include" "${OSV_COS_SOURCE}/third_party/include")
  target_link_libraries(osv-cpp-cos PRIVATE "${OSV_COS_LIBRARY}" nlohmann_json::nlohmann_json
    Poco::NetSSL Poco::Net Poco::Crypto Poco::Util Poco::JSON Poco::XML Poco::Foundation
    OpenSSL::SSL OpenSSL::Crypto Threads::Threads)
  target_compile_options(osv-cpp-cos PRIVATE -Wall -Wextra -Wpedantic)
endif()

# OBS is a C SDK linked by a C++ worker; keep its standalone dependency build intact.
set(OSV_OBS_SOURCE "" CACHE PATH "Official huaweicloud-sdk-c-obs checkout")
set(OSV_OBS_LIBRARY "" CACHE FILEPATH "libeSDKOBS.so built from OSV_OBS_SOURCE")
if(OSV_OBS_SOURCE OR OSV_OBS_LIBRARY)
  if(NOT OSV_OBS_SOURCE OR NOT EXISTS "${OSV_OBS_LIBRARY}")
    message(FATAL_ERROR "OBS requires both OSV_OBS_SOURCE and a built OSV_OBS_LIBRARY")
  endif()
  set(OSV_OBS_INCLUDE "${OSV_OBS_SOURCE}/source/eSDK_OBS_API/eSDK_OBS_API_C++/inc")
  file(STRINGS "${OSV_OBS_INCLUDE}/request.h" OSV_OBS_VERSION_LINE REGEX "^#define OBS_SDK_VERSION ")
  string(REGEX MATCH "[0-9]+\\.[0-9]+\\.[0-9]+" OSV_OBS_VERSION "${OSV_OBS_VERSION_LINE}")
  file(STRINGS "${OSV_OBS_INCLUDE}/request.h" OSV_OBS_ATTEMPTS_LINE REGEX "^#define RETRY_NUM ")
  string(REGEX MATCH "[0-9]+" OSV_OBS_ATTEMPTS "${OSV_OBS_ATTEMPTS_LINE}")
  if(NOT OSV_OBS_VERSION OR NOT OSV_OBS_ATTEMPTS)
    message(FATAL_ERROR "Cannot identify the supplied OBS SDK version and retry loop budget")
  endif()
  set(OSV_OBS_REVISION "unknown")
  find_package(Git QUIET)
  if(Git_FOUND AND EXISTS "${OSV_OBS_SOURCE}/.git")
    execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${OSV_OBS_SOURCE}" rev-parse HEAD
      OUTPUT_VARIABLE OSV_OBS_REVISION OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
  endif()
  add_executable(osv-cpp-obs obs_worker.cpp)
  target_include_directories(osv-cpp-obs PRIVATE "${OSV_OBS_INCLUDE}")
  target_compile_definitions(osv-cpp-obs PRIVATE OSV_OBS_SDK_VERSION="${OSV_OBS_VERSION}"
    OSV_OBS_SOURCE_REVISION="${OSV_OBS_REVISION}" OSV_OBS_ATTEMPTS=${OSV_OBS_ATTEMPTS})
  target_link_libraries(osv-cpp-obs PRIVATE "${OSV_OBS_LIBRARY}" nlohmann_json::nlohmann_json)
  if(UNIX AND NOT APPLE)
    get_filename_component(OSV_OBS_LIBRARY_DIR "${OSV_OBS_LIBRARY}" DIRECTORY)
    target_link_options(osv-cpp-obs PRIVATE "-Wl,-rpath-link,${OSV_OBS_LIBRARY_DIR}")
  endif()
  target_compile_options(osv-cpp-obs PRIVATE -Wall -Wextra -Wpedantic)
endif()
